"""
09_knowledge_distillation.py

Feature-Guided Knowledge Distillation for EEG Seizure Detection

Pipeline:
    1. Load the existing 330-feature EEG feature table.
    2. Perform patient-independent Leave-One-Patient-Out evaluation.
    3. Train an XGBoost teacher on the four training patients.
    4. Use teacher feature importance to select top-K EEG features.
    5. Train a lightweight neural-network student:
         - baseline student: hard labels only
         - KD student: hard labels + teacher soft targets
    6. Evaluate both students on the completely held-out patient.
    7. Save metrics, predictions, selected features and presentation figures.

Important:
    - Feature selection is performed separately inside every LOPO fold.
    - The held-out patient is never used for feature selection or training.
    - This is a basic Review-2 implementation of feature-guided KD.
"""

STUDENT_HIDDEN_2 = 16

STUDENT_BATCH_SIZE = 256
STUDENT_LEARNING_RATE = 0.001
STUDENT_WEIGHT_DECAY = 1e-4

# Knowledge distillation

KD_ALPHA = 0.5

# Same probability threshold for fair initial comparison.
DECISION_THRESHOLD = 0.50

# ============================================================
# XGBOOST TEACHER
# ============================================================

def make_teacher(y_train):
    positive = int((y_train == 1).sum())
    negative = int((y_train == 0).sum())

    if positive == 0:
        raise ValueError("Training fold contains no seizure samples.")

    scale_pos_weight = negative / positive

    model = XGBClassifier(
        n_estimators=TEACHER_N_ESTIMATORS,
        max_depth=TEACHER_MAX_DEPTH,
        learning_rate=TEACHER_LEARNING_RATE,
        subsample=TEACHER_SUBSAMPLE,
        colsample_bytree=TEACHER_COLSAMPLE,
        scale_pos_weight=scale_pos_weight,
        objective="binary:logistic",
        eval_metric="logloss",
        tree_method="hist",
        random_state=RANDOM_STATE,
        n_jobs=-1,
    )

    return model


# ============================================================
# STUDENT MODEL
# ============================================================

class LightweightStudent(nn.Module):
    """
    Small MLP student model.

    Input:
        Top-K teacher-selected EEG features

    Output:
        One seizure logit
    """

    def __init__(self, input_dim):
        super().__init__()

        self.network = nn.Sequential(
            nn.Linear(input_dim, STUDENT_HIDDEN_1),
            nn.ReLU(),
            nn.Linear(STUDENT_HIDDEN_1, STUDENT_HIDDEN_2),
            nn.ReLU(),
            nn.Linear(STUDENT_HIDDEN_2, 1),
        )

    def forward(self, x):
        return self.network(x).squeeze(1)


# ============================================================
# FEATURE SCALING
# ============================================================

def fit_standard_scaler(X_train):
    mean = X_train.mean(axis=0)
    std = X_train.std(axis=0)

    std = np.where(std < 1e-8, 1.0, std)

    return mean, std


def apply_standard_scaler(X, mean, std):
    return (X - mean) / std


# ============================================================
# TEACHER FEATURE GUIDANCE
# ============================================================

def select_top_features(teacher, feature_names, k):
    importances = teacher.feature_importances_

    ranking = (
        pd.DataFrame({
            "feature": feature_names,
            "importance": importances,
        })
        .sort_values("importance", ascending=False)
        .reset_index(drop=True)
    )

    selected = ranking.head(min(k, len(ranking))).copy()

    return selected


# ============================================================
# TEACHER PROBABILITY → SOFT TARGET
# ============================================================

def probability_to_logit(probability):
    probability = np.clip(probability, 1e-6, 1 - 1e-6)
    return np.log(probability / (1.0 - probability))


def teacher_soft_targets(teacher_probability, temperature):
    """
    Convert teacher probabilities into temperature-softened
    binary probabilities.

    Higher temperature produces softer teacher targets.
    """

    teacher_logits = probability_to_logit(teacher_probability)

    softened = 1.0 / (
        1.0 + np.exp(-teacher_logits / temperature)
    )

    return softened.astype(np.float32)


# ============================================================
# TRAIN STUDENT
# ============================================================

def train_student(
    X_train,
    y_train,
    teacher_probability=None,
    use_kd=False,
):
    """
    Train either:

    1. Ordinary lightweight student:
       hard-label BCE only

    2. KD student:
       alpha * hard BCE
       +
       (1-alpha) * temperature-scaled soft-target loss
    """

    X_tensor = torch.tensor(
        X_train,
        dtype=torch.float32,
    )

    y_tensor = torch.tensor(
        y_train.astype(np.float32),
        dtype=torch.float32,
    )

    dataset = TensorDataset(
        X_tensor,
        y_tensor,
    )

    loader = DataLoader(
        dataset,
        batch_size=STUDENT_BATCH_SIZE,
        shuffle=True,
        drop_last=False,
    )

    model = LightweightStudent(
        input_dim=X_train.shape[1]
    ).to(DEVICE)

    optimizer = torch.optim.Adam(
        model.parameters(),
        lr=STUDENT_LEARNING_RATE,
        weight_decay=STUDENT_WEIGHT_DECAY,
    )

    positive = float((y_train == 1).sum())
    negative = float((y_train == 0).sum())

    if positive > 0:
        pos_weight_value = negative / positive
    else:
        pos_weight_value = 1.0

    pos_weight = torch.tensor(
        [pos_weight_value],
        dtype=torch.float32,
        device=DEVICE,
    )

    hard_loss_function = nn.BCEWithLogitsLoss(
        pos_weight=pos_weight
    )

    if use_kd:
        if teacher_probability is None:
            raise ValueError(
                "Teacher probabilities are required for KD training."
            )

        soft_targets = teacher_soft_targets(
            teacher_probability,
            KD_TEMPERATURE,
        )

        soft_tensor = torch.tensor(
            soft_targets,
            dtype=torch.float32,
        )

        dataset = TensorDataset(
            X_tensor,
            y_tensor,
            soft_tensor,
        )

        loader = DataLoader(
            dataset,
            batch_size=STUDENT_BATCH_SIZE,
            shuffle=True,
            drop_last=False,
        )

    model.train()

    for epoch in range(STUDENT_EPOCHS):

        epoch_loss = 0.0

        for batch in loader:

            optimizer.zero_grad()

            if use_kd:
                xb, yb, teacher_soft = batch

                xb = xb.to(DEVICE)
                yb = yb.to(DEVICE)
                teacher_soft = teacher_soft.to(DEVICE)

                student_logits = model(xb)

                hard_loss = hard_loss_function(
                    student_logits,
                    yb,
                )

                student_soft_logits = (
                    student_logits / KD_TEMPERATURE
                )

                kd_loss = F.binary_cross_entropy_with_logits(
                    student_soft_logits,
                    teacher_soft,
                )

                kd_loss = (
                    kd_loss
                    * (KD_TEMPERATURE ** 2)
                )

                loss = (
                    KD_ALPHA * hard_loss
                    +
                    (1.0 - KD_ALPHA) * kd_loss
                )

            else:

                xb, yb = batch

                xb = xb.to(DEVICE)
                yb = yb.to(DEVICE)

                student_logits = model(xb)

                loss = hard_loss_function(
                    student_logits,
                    yb,
                )

            loss.backward()
            optimizer.step()

            epoch_loss += loss.item()

        if (
            epoch == 0
            or epoch == STUDENT_EPOCHS - 1
        ):
            print(
                f"      Epoch {epoch + 1:02d}/"
                f"{STUDENT_EPOCHS} "
                f"loss={epoch_loss / len(loader):.4f}"
            )

    return model


# ============================================================
# STUDENT PREDICTION
# ============================================================

def predict_student(model, X):
    model.eval()

    X_tensor = torch.tensor(
        X,
        dtype=torch.float32,
    ).to(DEVICE)

    with torch.no_grad():
        logits = model(X_tensor)
        probability = torch.sigmoid(logits)

    return probability.cpu().numpy()


# ============================================================
# METRICS
# ============================================================

def calculate_metrics(y_true, probability):
    prediction = (
        probability >= DECISION_THRESHOLD
    ).astype(int)

    metrics = {
        "accuracy": accuracy_score(
            y_true,
            prediction,
        ),
        "precision": precision_score(
            y_true,
            prediction,
            zero_division=0,
        ),
        "recall_sensitivity": recall_score(
            y_true,
            prediction,
            zero_division=0,
        ),
        "f1": f1_score(
            y_true,
            prediction,
            zero_division=0,
        ),
    }

    if len(np.unique(y_true)) == 2:
        metrics["roc_auc"] = roc_auc_score(
            y_true,
            probability,
        )
    else:
        metrics["roc_auc"] = np.nan

    return metrics, prediction


# ============================================================
# PARAMETER COUNT
# ============================================================

def count_parameters(model):
    return sum(
        p.numel()
        for p in model.parameters()
        if p.requires_grad
    )
