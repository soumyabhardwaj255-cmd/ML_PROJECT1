# IDENTIFY FEATURE / IMPORTANCE COLUMNS
# ============================================================

def identify_columns(df):
    feature_candidates = [
        "feature",
        "Feature",
        "feature_name",
        "Feature_Name",
    ]

    importance_candidates = [
        "importance",
        "Importance",
        "feature_importance",
        "Feature_Importance",
    ]

    feature_col = next(
        (c for c in feature_candidates if c in df.columns),
        None
    )

    importance_col = next(
        (c for c in importance_candidates if c in df.columns),
        None
    )

    if feature_col is None or importance_col is None:
        raise ValueError(
            f"Could not identify feature/importance columns.\n"
            f"Available columns: {list(df.columns)}"
        )

    return feature_col, importance_col


rf_feature_col, rf_importance_col = identify_columns(rf)
xgb_feature_col, xgb_importance_col = identify_columns(xgb)

)

# ============================================================
# FEATURE FAMILY EXTRACTION
# ============================================================

def feature_family(feature_name):
    """
    Extract the broad feature family from our feature naming scheme.

    Examples:
        time__std__C3-P3       -> time
        frequency__delta__C3-P3 -> frequency
        entropy__spectral__C3-P3 -> entropy
        hjorth__mobility__C3-P3 -> hjorth
    """
    parts = str(feature_name).split("__")

    if len(parts) >= 1:
        return parts[0]

    return "unknown"


def feature_channel(feature_name):
    """
    Extract EEG channel from our feature naming scheme.
    """
    parts = str(feature_name).split("__")

    if len(parts) >= 3:
        return parts[-1]

    return "unknown"

rf["channel"] = rf["feature"].apply(feature_channel)

xgb["family"] = xgb["feature"].apply(feature_family)
xgb["channel"] = xgb["feature"].apply(feature_channel)
